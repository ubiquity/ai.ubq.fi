def tokens(s):
 out=[];i=0
 while i<len(s):
  if s[i].isspace():i+=1;continue
  if s.startswith('//',i):
   j=s.find('\n',i);i=len(s) if j<0 else j+1;continue
  if s.startswith('/*',i):
   depth=1;i+=2
   while depth:
    if s.startswith('/*',i):depth+=1;i+=2
    elif s.startswith('*/',i):depth-=1;i+=2
    else:i+=1
   continue
  raw=re.match(r'(?:br|cr|r)(#*)"',s[i:])
  if raw:
   close='"'+raw.group(1);end=s.index(close,i+len(raw.group(0)))+len(close);out.append(s[i:end]);i=end;continue
  quoted=re.match(r'(?:b|c)?"',s[i:])
  char=re.match(r"(?:b)?'(?:\\(?:u\{[0-9a-fA-F_]+\}|x[0-9a-fA-F]{2}|.)|[^\\'\n])'",s[i:])
  if quoted:
   end=i+len(quoted.group(0))
   while end<len(s):
    if s[end]=='\\':end+=2
    elif s[end]=='"':end+=1;break
    else:end+=1
   out.append(s[i:end]);i=end;continue
  if char:out.append(char.group());i+=len(char.group());continue
  word=re.match(r'(?:r#)?[A-Za-z_][A-Za-z0-9_]*|[0-9]+(?:\.[0-9]+)?(?:[a-z][0-9]+)?',s[i:])
  if word:out.append(word.group());i+=len(word.group());continue
  out.append(s[i]);i+=1
 return out
