cmake_minimum_required(VERSION 3.25)
project(uos_cpp_pilot VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_SKIP_RPATH ON)
set(CMAKE_BUILD_RPATH "")
set(CMAKE_INSTALL_RPATH "")
set(Boost_USE_STATIC_LIBS ON)
find_package(Boost 1.81.0 EXACT CONFIG REQUIRED COMPONENTS json system)
find_package(OpenSSL 3 REQUIRED)
find_package(Threads REQUIRED)
if(NOT CMAKE_CROSSCOMPILING)
  foreach(target Boost::headers Boost::json Boost::system Boost::container)
    if(TARGET ${target})
      set_property(TARGET ${target} PROPERTY INTERFACE_INCLUDE_DIRECTORIES "$ENV{UOS_CPP_TOOLCHAIN_DIR}/host-tools/include")
    endif()
  endforeach()
endif()
add_library(pilot_core src/core.cpp src/project_input.cpp src/project_tools.cpp src/payload_policy.cpp src/translator.cpp src/fixture_state.cpp src/fixture_control.cpp)
target_include_directories(pilot_core PUBLIC include)
target_link_libraries(pilot_core PUBLIC Boost::json Boost::system OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
add_executable(uos_cpp_pilot src/main.cpp src/server.cpp src/session.cpp src/upstream.cpp src/transport.cpp)
target_link_libraries(uos_cpp_pilot PRIVATE pilot_core)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}/.." OUTPUT_VARIABLE UOS_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE UOS_GIT_RESULT)
execute_process(COMMAND git status --porcelain --untracked-files=normal -- cpp docs/DECISIONS.md WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}/.." OUTPUT_VARIABLE UOS_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE)
if(UOS_GIT_RESULT EQUAL 0 AND UOS_DIRTY STREQUAL "")
  set(UOS_IDENTITY_SHA "${UOS_COMMIT}")
  set(UOS_IDENTITY_DEPLOYMENT "cpp-${UOS_COMMIT}")
  set(UOS_IDENTITY_DIRTY 0)
else()
  set(UOS_IDENTITY_SHA "unknown")
  set(UOS_IDENTITY_DEPLOYMENT "cpp-dirty")
  set(UOS_IDENTITY_DIRTY 1)
endif()
target_compile_definitions(uos_cpp_pilot PRIVATE UOS_GIT_SHA="${UOS_IDENTITY_SHA}" UOS_DEPLOYMENT_ID="${UOS_IDENTITY_DEPLOYMENT}" UOS_BUILD_DIRTY=${UOS_IDENTITY_DIRTY})

foreach(target pilot_core uos_cpp_pilot)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror=return-type -ffile-prefix-map=${PROJECT_SOURCE_DIR}=uos-cpp-pilot -ffile-prefix-map=$ENV{UOS_CPP_TOOLCHAIN_DIR}=uos-sdk -fdebug-prefix-map=${CMAKE_BINARY_DIR}=uos-cpp-build)
  if(UOS_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  endif()
endforeach()
install(TARGETS uos_cpp_pilot RUNTIME DESTINATION bin)
enable_testing()
add_executable(protocol_tests tests/protocol_tests.cpp)
target_link_libraries(protocol_tests PRIVATE pilot_core)
option(UOS_ANALYSIS_CONTROLS "Expose analysis-only fixture compilation commands" OFF)
if(UOS_ANALYSIS_CONTROLS)
  add_library(analysis_controls OBJECT EXCLUDE_FROM_ALL tests/analysis/raw_defects.cpp tests/analysis/clean_ownership.cpp tests/analysis/clean_cancel.cpp)
  target_link_libraries(analysis_controls PRIVATE pilot_core)
  target_include_directories(analysis_controls SYSTEM PRIVATE tests/analysis/fixture_sdk)
  add_custom_target(cpp-static-analysis
    COMMAND /usr/bin/python3 "${PROJECT_SOURCE_DIR}/tools/analyze.py" "$ENV{UOS_CPP_TOOLCHAIN_DIR}" pi-release
    WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}/.."
    VERBATIM)
endif()
if(UOS_SANITIZE)
  target_compile_options(protocol_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  target_link_options(protocol_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
endif()
add_test(NAME protocol COMMAND protocol_tests)
set_tests_properties(protocol PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
