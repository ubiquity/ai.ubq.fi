# Stable integration API: each source is compiled once, and test closures select
# individual objects instead of importing unrelated registrars or definitions.
function(uos_object_for_source output source)
  if(IS_ABSOLUTE "${source}" OR "${source}" MATCHES "(^|/)\\.\\.(/|$)")
    message(FATAL_ERROR "Object source must be relative to cpp/gateway: ${source}")
  endif()
  string(SHA256 identity "${source}")
  string(SUBSTRING "${identity}" 0 16 identity)
  set(target "uos_obj_${identity}")
  if(NOT TARGET "${target}")
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
      message(FATAL_ERROR "Missing object source: ${source}")
    endif()
    add_library("${target}" OBJECT EXCLUDE_FROM_ALL "${source}")
    target_link_libraries("${target}" PRIVATE uos_gateway_settings)
    if(UOS_GATEWAY_GIT_COMMIT AND source IN_LIST UOS_PACKAGE_IDENTITY_SOURCES)
      # The commit changes with every package build. Keep it out of every other
      # object so their compiler-cache keys stay stable.
      target_compile_definitions("${target}" PRIVATE "UOS_PACKAGE_GIT_COMMIT=\"${UOS_GATEWAY_GIT_COMMIT}\"")
    endif()
    get_property(simdjson_consumers GLOBAL PROPERTY UOS_SIMDJSON_CONSUMERS)
    if(source IN_LIST simdjson_consumers)
      target_compile_definitions("${target}" PRIVATE UOS_GATEWAY_SIMDJSON)
      target_include_directories("${target}" SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/simdjson")
    endif()
    foreach(vendor IN ITEMS uos_vendor_qcbor uos_vendor_quickjs-libregexp)
      if(TARGET "${vendor}")
        get_property(consumers TARGET "${vendor}" PROPERTY UOS_VENDOR_CONSUMERS)
        if(source IN_LIST consumers)
          target_link_libraries("${target}" PRIVATE "${vendor}")
        endif()
      endif()
    endforeach()
  endif()
  set("${output}" "${target}" PARENT_SCOPE)
endfunction()

set(UOS_GATEWAY_OBJECTS)
foreach(source IN LISTS UOS_ALL_SOURCES)
  uos_object_for_source(object "${source}")
  list(APPEND UOS_GATEWAY_OBJECTS "$<TARGET_OBJECTS:${object}>")
endforeach()

function(uos_add_gateway_executable target entry)
  add_executable("${target}" "${entry}")
  target_link_libraries("${target}" PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,uos_gateway_modules>")
endfunction()
uos_add_gateway_executable(uos_gateway entrypoints/gateway.cpp)
